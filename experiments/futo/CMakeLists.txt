cmake_minimum_required(VERSION 3.22)
project(futo_probe LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_INSTALL_INCDIR ${CMAKE_INSTALL_INCLUDEDIR})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED FUTO_SOURCE OR NOT DEFINED FUTO_PREPARED)
  message(FATAL_ERROR "Set FUTO_SOURCE and FUTO_PREPARED to locally prepared sources")
endif()
set(SPM_USE_BUILTIN_PROTOBUF ON)
set(SPM_ENABLE_SHARED OFF)
set(SPM_ENABLE_NFKC_COMPILE OFF)
add_subdirectory(${FUTO_SOURCE}/src/sentencepiece sentencepiece EXCLUDE_FROM_ALL)
target_include_directories(sentencepiece-static PRIVATE ${FUTO_SOURCE}/src ${FUTO_SOURCE}/src/sentencepiece)
set(G ${FUTO_SOURCE}/src/ggml)
add_executable(futo-probe probe.cpp
  ${G}/ggml.c ${G}/ggml-alloc.c ${G}/ggml-backend.c ${G}/ggml-quants.c
  ${G}/llama.cpp ${G}/LanguageModel.cpp ${G}/ModelMeta.cpp ${G}/context.cpp)
target_include_directories(futo-probe PRIVATE ${FUTO_PREPARED} ${FUTO_SOURCE}/src ${G}
  ${FUTO_SOURCE}/src/sentencepiece ${FUTO_SOURCE}/src/third_party)
target_link_libraries(futo-probe PRIVATE sentencepiece-static)
if(MSVC)
  target_compile_options(futo-probe PRIVATE /utf-8 /arch:AVX2)
  target_compile_definitions(futo-probe PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(futo-probe PRIVATE -O3)
endif()
